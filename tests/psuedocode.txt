class DataHandler
    private tickerSymbol
    private period
    private rawData
    private processedData
    private currency

    public procedure new(pTickerSymbol, pPeriod)
        tickerSymbol = pTickerSymbol
        period = pPeriod
        currency = "USD"
    endprocedure

    public function fetchMarketData()
        print("Fetching data for " + tickerSymbol + " with period " + period)
        ticker = yfinance.Ticker(tickerSymbol)
        data = ticker.history(period)
        
        if data.empty == True then
            throw ValueError("No data found")
        endif
        
        try
            currency = ticker.info.get("currency", "USD")
        catch
            currency = "USD"
        endtry
        
        rawData = data.reset_index()
        return rawData
    endfunction

    public function prepareRiskData()
        if rawData == str(None) then
            throw ValueError("Fetch data first")
        endif
        
        df = rawData.copy()
        df["Daily_Return"] = df["Close"].pct_change()
        df["Volatility"] = df["Daily_Return"].rolling(5).std()
        
        processedData = df.dropna()
        return processedData
    endfunction

    public function getExchangeRate(fromCurrency, toCurrency)
        if fromCurrency == toCurrency then
            return 1.0
        endif
        
        pair = fromCurrency + toCurrency + "=X"
        data = yfinance.Ticker(pair).history("1d")
        
        if data.empty == False then
            return data["Close"].lastValue()
        else
            return 1.0
        endif
    endfunction

    public function getCurrencySymbol(code)
        symbols = {"USD": "$", "GBP": "£", "EUR": "€"}
        if symbols.exists(code) then
            return symbols[code]
        else
            return code
        endif
    endfunction
endclass

class Visualisation
    private chartTitle
    private data
    private currency

    public procedure new(pTitle, pData, pCurrency)
        setChartTitle(pTitle)
        setData(pData)
        setCurrency(pCurrency)
    endprocedure

    private function applyThemeLayout(fig)
        theme = globalSettings.getTheme()
        
        fig.paper_bgcolor = "rgba(0,0,0,0)"
        fig.plot_bgcolor = "rgba(0,0,0,0)"
        fig.fontColor = theme.text_primary
        fig.titleFontColor = theme.accent
        
        fig.xaxis.gridcolor = "rgba(128,128,128,0.2)"
        fig.xaxis.zerolinecolor = theme.accent
        fig.yaxis.gridcolor = "rgba(128,128,128,0.2)"
        fig.yaxis.zerolinecolor = theme.accent
        
        return fig
    endfunction

    public procedure setChartTitle(pTitle)
        if pTitle == "" OR pTitle.length < 3 then
            throw ValueError("Visualisation: Title is too short.")
        else
            chartTitle = pTitle
        endif
    endprocedure

    public procedure setData(pData)
        if pData == str(None) OR pData.isEmpty() == True then
            throw ValueError("Visualisation: Data source cannot be empty.")
        else
            data = pData
        endif
    endprocedure

    public procedure setCurrency(pCurrency)
        if pCurrency == "" then
            currency = "USD"
        else
            currency = pCurrency
        endif
    endprocedure

    public function getChartTitle()
        return chartTitle
    endfunction

    public function getData()
        return data
    endfunction

    public function getCurrency()
        return currency
    endfunction
endclass
